Table Accounts {
  account_id int [primary key]
  username varchar
  email varchar
  password varchar
  profile_pic varchar
  account_banned boolean
}

Table Friends {
  account1_id int [ref: > Accounts.account_id]
  account2_id int [ref: > Accounts.account_id]
  friends_since date
}

Table Chats {
  chat_id int [primary key]
  chat_name varchar
  chat_image varchar
}

Table Messages {
  message_id int [primary key]
  chat_id int [ref: > Chats.chat_id]
  sent_by int [ref: > Accounts.account_id]
  message text
}

Table User_in_Chat {
  account_id int [ref: > Accounts.account_id]
  chat_id int [ref: > Chats.chat_id]
  role varchar
}

Table Games {
  game_id int [primary key]
  name varchar
  description text
  multiplayer boolean
  system_req_id int [ref: < Hardware.hardware_id]
  publisher varchar
  release_date date
}

Table System_Requirements {
  system_req_id int [primary key]
  cpu varchar
  memory varchar
  gpu varchar
}

Table Game_Bans {
  ban_id int [primary key]
  account_id int [ref: > Accounts.account_id]
  game_id int [ref: > Games.game_id]
  expires date
  reason text
}

Table Ban_Reasons {
  ban_id int [ref: > Game_Bans.ban_id]
  reason text
}

Table Owned_Games {
  account_id int [ref: > Accounts.account_id]
  game_id int [ref: > Games.game_id]
  date_acquired date
}

Table Reviews {
  review_id int [primary key]
  written_by int [ref: > Accounts.account_id]
  game_id int [ref: > Games.game_id]
  text text
  rating int
}

Table Platform {
  hardware_id int [ref: < Hardware.hardware_id]
  mac boolean
  windows boolean
  linux boolean
}

Table Hardware {
  hardware_id int [primary key]
  min_re varchar [ref: < System_Requirements.system_req_id]
  rec_re varchar [ref: < System_Requirements.system_req_id]
  storage varchar
  vr boolean
}
